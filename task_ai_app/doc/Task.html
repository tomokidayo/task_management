def hhmm_to_minutes(value)
  return value if value.is_a?(Integer)
  return nil if value.blank?
  h, m = value.split(":").map(&:to_i)
  h * 60 + m
end

def minutes_to_hhmm(minutes)
  return nil if minutes.nil?
  format("%02d:%02d", minutes / 60, minutes % 60)
end
